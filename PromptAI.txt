  BASE = r"C:\Z_Srinu\WorkingDat\StocksDatabase\NSE Data\Day\market_data"

  def load_index():
      path = os.path.join(BASE, "symbol=NIFTYMIDSMALLCAP400")
      files = glob.glob(os.path.join(path, "*.parquet"))
      df = pd.concat([pd.read_parquet(f) for f in sorted(files)])
      df["date"] = pd.to_datetime(df["eodtime"], format="%d-%b-%Y")
      return df.sort_values("date").reset_index(drop=True)

  def load_stock(symbol):
      path = os.path.join(BASE, f"symbol={symbol}")
      files = glob.glob(os.path.join(path, "*.parquet"))
      df = pd.concat([pd.read_parquet(f) for f in sorted(files)])
      df["date"] = pd.to_datetime(df["Date"], format="%d-%b-%Y")
      return df.sort_values("date").reset_index(drop=True)

  def load_symbols():
